{{- /* Language switcher used both in the site header (layouts/partials/header.html) and directly under the name on the profile-mode home page (layouts/partials/index_profile.html). Unlike the original PaperMod behaviour (a single link to "the other" language), this always shows BOTH languages, in a fixed order — French first, then English — and marks the active one with the same "active" look as the main navigation (see .lang-switch .active in header.css), instead of it being a plain, easy-to-miss link. Expects "." to be the current Page (so .Lang works). Always call it as: {{ partial "lang_switch.html" . }} or, from inside a "with" block where "." has changed, capture the page first: {{- $page := . -}} ... {{ partial "lang_switch.html" $page }} */ -}} {{- $current := .Lang -}} {{- $frPage := "" -}} {{- $enPage := "" -}} {{- range site.Home.AllTranslations }} {{- if eq .Lang "fr" }}{{ $frPage = . }}{{ end }} {{- if eq .Lang "en" }}{{ $enPage = . }}{{ end }} {{- end }}
    {{- with $frPage }}
  • {{- if eq $current "fr" }} {{ i18n "frenchLabel" }} {{- else }} {{ i18n "frenchLabel" }} {{- end }}
    • {{- end }} {{- if and $frPage $enPage }}
  • /
    • {{- end }} {{- with $enPage }}
  • {{- if eq $current "en" }} {{ i18n "englishLabel" }} {{- else }} {{ i18n "englishLabel" }} {{- end }}
    • {{- end }}
